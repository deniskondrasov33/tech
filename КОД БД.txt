CREATE DATABASE tech_shop_1
--  1.Таблица Магазины
CREATE TABLE shops (
   KPP  INT PRIMARY KEY,
    Name VARCHAR(255),
    Adress VARCHAR(255),
    INN BIGINT,
    OKPO INT
);
-- 2.Таблица  Группы товаров
CREATE TABLE product_group (
    OKPD_2 INT  PRIMARY KEY,
    Group_name VARCHAR(255)
);

-- 3.Таблица Товары
CREATE TABLE products (
    GTIN BIGINT  PRIMARY KEY,
    Name VARCHAR(255),
    OKPD_2 INT,
    TN_VED VARCHAR(255),
    Characteristics VARCHAR(255),
    Price DECIMAL(10,2),

    FOREIGN KEY (OKPD_2)
        REFERENCES product_group(OKPD_2)
);

--  4.Таблица Маркированные единицы
CREATE TABLE marked_units (
    Marking_code BIGINT PRIMARY KEY,
    serial_number VARCHAR(255),
    GTIN BIGINT,

    FOREIGN KEY (GTIN)
        REFERENCES products(GTIN)
);
 -- 5. Таблица  Клиенты
CREATE TABLE clients (
    id_card INT  PRIMARY KEY,
    Full_name VARCHAR(255),
    Bonus_account BIGINT
);
 -- 6. Таблица Продажи
CREATE TABLE sales (
    sale_code BIGINT NOT NULL PRIMARY KEY AUTO_INCREMENT,
    KPP INT ,
    Marking_code BIGINT ,
    date_and_time DATETIME ,
    payment_method VARCHAR(255),
    `check` VARCHAR(255),
    client_ID INT,
    
    FOREIGN KEY (KPP)
        REFERENCES shops(KPP),

    FOREIGN KEY (Marking_code)
        REFERENCES marked_units(Marking_code),

    FOREIGN KEY (client_ID)
        REFERENCES clients(id_card)
);
 
CREATE TABLE returns (
    return_number INT  PRIMARY KEY AUTO_INCREMENT,
    sale_code BIGINT NOT NULL,
    date_return DATE,
    Reason TEXT,
    Passport  VARCHAR(255),
    FOREIGN KEY (    sale_code)
    REFERENCES sales (sale_code)
);
 -- 8. Таблица  Заявки на перемещение
CREATE TABLE transfer (
    transfer_number  INT  PRIMARY KEY,
    KPP_sender INT,
    KPP_receiver INT,
    Status  VARCHAR(255),

    FOREIGN KEY (KPP_sender)
        REFERENCES shops(KPP),

    FOREIGN KEY (KPP_receiver)
        REFERENCES shops(KPP)
);
 -- 9. Таблица  Позиции заявок на перемещение
CREATE TABLE transfer_positions (
    Position_number_in_transfer  INT ,
    transfer_number  INT ,
    Marking_code BIGINT,

    PRIMARY KEY (
        Position_number_in_transfer,
        transfer_number
    ),

    FOREIGN KEY (transfer_number )
        REFERENCES transfer (    transfer_number  ),

    FOREIGN KEY (Marking_code)
        REFERENCES marked_units (Marking_code)
);

-- Изменение структуры таблиц с помощью

ALTER TABLE transfer

ALTER TABLE transfer_positions
    MODIFY Position_number_in_transfer INT NOT NULL,
    MODIFY transfer_number INT NOT NULL;

-- Лабораторная 2

--  Просмотр оформленных возвратов (старший продавец)
SELECT * FROM sales, returns
WHERE sales.sale_code = returns.sale_code;

--  Просмотр товаров и групп (старший продавец)
SELECT * FROM products, product_group
WHERE products.OKPD_2 = product_group.OKPD_2
ORDER BY products.Price DESC;

--  Просмотр заявок и их позиций (старший продавец)
SELECT *FROM transfer_positions
INNER JOIN transfer ON transfer.transfer_number = transfer_positions.transfer_number;

--  Поиск товаров без возврата (старший продавец)
SELECT sales.Marking_code,
    products.Name
FROM sales
LEFT JOIN returns
    ON sales.sale_code = returns.sale_code
JOIN marked_units
    ON sales.Marking_code = marked_units.Marking_code
JOIN products
    ON marked_units.GTIN = products.GTIN
WHERE returns.sale_code IS NULL;

--  Расчёт суммы продаж за день (менеджер торговой сети)
SELECT sum(products.price) AS sumprod
FROM products 
LEFT JOIN marked_units ON marked_units.GTIN = products.GTIN
INNER JOIN sales ON sales.Marking_code = marked_units.Marking_code
WHERE date_and_time >= '2026-10-26 00:00:00' AND date_and_time < '2026-10-27 00:00:00';

--  Расчёт средней цены продаж (менеджер торговой сети)
SELECT AVG(products.price) AS avgprod
FROM products 
LEFT JOIN marked_units ON marked_units.GTIN = products.GTIN
INNER JOIN sales ON sales.Marking_code = marked_units.Marking_code
WHERE date_and_time >= '2026-10-26 00:00:00' AND date_and_time < '2026-10-27 00:00:00';


--  Подсчёт продаж магазина (менеджер торговой сети)
SELECT COUNT(*) AS All_sales
FROM sales
WHERE sales.KPP = 1001;



--  Поиск часто возвращаемых товаров (менеджер торговой сети)
SELECT products.Name
FROM products 
INNER JOIN marked_units ON marked_units.GTIN  =  products.GTIN
INNER JOIN sales ON marked_units.Marking_code = sales.Marking_code
INNER JOIN returns ON returns.sale_code = sales.sale_code
GROUP BY products.Name
HAVING COUNT(returns.return_number) > 3;

-- Лабораторная работа 3

--  Поиск покупателей магазина (директор магазина)
SELECT clients.Full_name FROM clients
WHERE clients.id_card IN (SELECT sales.client_ID FROM sales WHERE sales.KPP = 1001);

--  Сравнение цен со смартфонами 
SELECT products.Name FROM products 
WHERE products.price > ALL(SELECT products.price FROM products
                           INNER JOIN product_group ON product_group.OKPD_2 = products.OKPD_2
                           WHERE product_group.Group_name = 'Смартфоны');

--  Разделение товаров по категориям
SELECT products.Name, products.Price,
       CASE WHEN products.Price > 60000.00  THEN 'Дорогой'
            WHEN products.Price > 30000.00 AND products.Price <= 60000.00  THEN 'Средний ценовой сегмент'
            WHEN products.Price <= 30000.00  THEN 'Бюджетный'
       END AS category
FROM products;

--  Определение состояния заявки (старший продавец)
SELECT transfer.transfer_number, IF(transfer.Status = 'Выполнена', 'Заявка выполнена', 'Заявка в работе') AS result
FROM transfer;

-- Представления для запроса о самых возвращаемых товарах
CREATE VIEW most_returned
AS SELECT products.Name
FROM products 
INNER JOIN marked_units ON marked_units.GTIN  =  products.GTIN
INNER JOIN sales ON marked_units.Marking_code = sales.Marking_code
INNER JOIN returns ON returns.sale_code = sales.sale_code
GROUP BY products.Name
HAVING COUNT(returns.return_number) > 3;

--  Поиск товаров с продажами выше среднего созданием временного представления (менеджер торговой сети)
WITH sales_count as (
    SELECT products.GTIN, products.Name, COUNT(*) AS Sale_amount
    FROM sales
    JOIN marked_units ON marked_units.Marking_code = sales.Marking_code
    JOIN products ON products.GTIN = marked_units.GTIN
    WHERE sales.KPP = 1001 GROUP BY products.GTIN)
SELECT * FROM sales_count
WHERE Sale_amount > (SELECT AVG(Sale_amount) FROM sales_count);
